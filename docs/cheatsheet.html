查询的骨架与执行顺序

SELECT   category, count(*) AS n          -- 5. 选出要的列、算表达式
FROM     shop.products p                  -- 1. 从哪些表
JOIN     shop.orders o ON o.product_id = p.id
WHERE    p.price > 10                     -- 2. 过滤行
GROUP BY category                         -- 3. 分组
HAVING   count(*) > 1                     -- 4. 过滤分组
ORDER BY n DESC                           -- 6. 排序
LIMIT    10;                              -- 7. 截取

连接、子查询、CTE、窗口函数

-- 内连接：两边都匹配的行
SELECT p.name, o.qty FROM shop.products p JOIN shop.orders o ON o.product_id = p.id;
-- 左连接：左表全部保留，右边没有匹配就是 NULL
SELECT p.name, o.qty FROM shop.products p LEFT JOIN shop.orders o ON o.product_id = p.id;
-- 反连接：从没被下过单的商品
SELECT p.name FROM shop.products p
WHERE NOT EXISTS (SELECT 1 FROM shop.orders o WHERE o.product_id = p.id);

-- CTE：给中间结果起个名字
WITH t AS (SELECT product_id, sum(qty) AS n FROM shop.orders GROUP BY product_id)
SELECT * FROM t WHERE n > 1;

-- 递归 CTE：生成 1..5（层级、树、图都用这个模式）
WITH RECURSIVE nums(n) AS (
  SELECT 1
  UNION ALL
  SELECT n + 1 FROM nums WHERE n < 5
)
SELECT n FROM nums;

-- 窗口函数：不合并行，在“窗口”里计算
SELECT name, price,
       rank() OVER (ORDER BY price DESC),                                  -- 排名
       sum(price) OVER (ORDER BY id ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW),  -- 累计
       lag(price) OVER (ORDER BY id)                                        -- 上一行的值
FROM shop.products;

-- 条件聚合
SELECT count(*) FILTER (WHERE price > 1) FROM shop.products;

建表、改表、写数据

CREATE SCHEMA shop;
CREATE TABLE shop.products (
  id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,   -- 自增主键（推荐写法）
  sku         text NOT NULL UNIQUE,
  name        text NOT NULL,
  price       numeric(10,2) NOT NULL CHECK (price > 0),
  attrs       jsonb NOT NULL DEFAULT '{}',
  tags        text[] NOT NULL DEFAULT '{}',
  created_at  timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE shop.orders (
  id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  product_id  bigint NOT NULL REFERENCES shop.products ON DELETE CASCADE,  -- 外键
  qty         int NOT NULL CHECK (qty > 0)
);

ALTER TABLE shop.products ADD COLUMN stock int NOT NULL DEFAULT 0;
ALTER TABLE shop.products ADD CONSTRAINT stock_nonneg CHECK (stock >= 0);
ALTER TABLE shop.products ALTER COLUMN name SET DATA TYPE varchar(200);
ALTER TABLE shop.products RENAME COLUMN stock TO stock_qty;
ALTER TABLE shop.products DROP CONSTRAINT stock_nonneg;

INSERT INTO shop.products (sku, name, price) VALUES ('A1', 'Latte', 4.5) RETURNING id;
UPDATE shop.products SET price = price * 1.1 WHERE sku = 'A1' RETURNING price;
INSERT INTO shop.products (sku, name, price) VALUES ('A1', 'Latte', 5)
  ON CONFLICT (sku) DO UPDATE SET price = EXCLUDED.price;                -- UPSERT
TRUNCATE shop.orders;                                                    -- 清空整表（很快，不可按条件）

CREATE VIEW shop.v AS SELECT name FROM shop.products;
CREATE MATERIALIZED VIEW shop.mv AS SELECT count(*) FROM shop.products;
REFRESH MATERIALIZED VIEW shop.mv;
常用类型说明
int / bigint整数；主键用 bigint 不怕不够用
numeric(p,s)精确小数，金额必用；不要用 float
text / varchar(n)字符串；Postgres 里两者性能一样，text 最常用
timestamptz带时区的时间戳，几乎总是选它而不是 timestamp
date / interval日期 / 时间间隔，如 interval '7 days'
boolean / uuid布尔 / UUID（gen_random_uuid() 生成）
jsonb二进制 JSON，可建 GIN 索引（第 5 章）
text[]、tstzrange、ENUM数组、范围、枚举（专题 2）
tsvector / vector(n) / geometry全文搜索 / pgvector 向量 / PostGIS 几何（第 6、8、10 章）

JSONB 与数组

写法结果写法结果
attrs -> 'milk'取字段，返回 jsonbattrs ->> 'size'取字段，返回 text
attrs #> '{milk,type}'按路径取，返回 jsonbattrs #>> '{milk,type}'按路径取，返回 text
attrs @> '{"size":"L"}'包含（可走 GIN）attrs ? 'size'有这个键
attrs ?| ARRAY['a','b']有任意一个键attrs ?& ARRAY['a','b']所有键都有
attrs || '{"hot":true}'合并（覆盖同名键）attrs - 'size'删掉一个键
jsonb_set(attrs, '{milk,type}', '"whole"')改某个路径的值jsonb_path_query(attrs, '$.sizes[*]')JSONPath 查询
jsonb_array_elements_text(attrs->'sizes')数组展开成行jsonb_build_object('a', 1) / jsonb_agg(x)构造 / 聚合成 JSON

全文搜索

SELECT to_tsvector('english', 'The quick brown foxes') @@ to_tsquery('english', 'fox & quick');  -- t
SELECT plainto_tsquery('english', 'quick foxes');                  -- 'quick' & 'fox'
SELECT phraseto_tsquery('english', 'brown fox');                   -- 'brown' <-> 'fox'（相邻）
SELECT websearch_to_tsquery('english', '"brown fox" -dog or cat'); -- 搜索框语法
SELECT ts_rank(to_tsvector('english', 'quick fox'), to_tsquery('fox'));        -- 相关度
SELECT ts_headline('english', 'The quick brown fox', to_tsquery('fox'));       -- 高亮 <b>fox</b>
SELECT setweight(to_tsvector('english', 'Title'), 'A') ||
       setweight(to_tsvector('english', 'body text'), 'B');                    -- 标题权重更高

事务与锁

BEGIN;
SAVEPOINT s1;
UPDATE shop.products SET price = 1;
ROLLBACK TO SAVEPOINT s1;       -- 只撤销 s1 之后的修改
COMMIT;

BEGIN ISOLATION LEVEL SERIALIZABLE;   -- 失败时 SQLSTATE 40001，要整个事务重试
…
COMMIT;

SELECT * FROM shop.products WHERE id = 1 FOR UPDATE;              -- 锁住，别人要等
SELECT * FROM shop.products WHERE id = 1 FOR UPDATE NOWAIT;       -- 锁不到立刻报错
SELECT * FROM shop.products WHERE id = 1 FOR UPDATE SKIP LOCKED;  -- 跳过被锁的行（任务队列）
SET lock_timeout = '2s';

# basic.py
import os
import psycopg
from psycopg.rows import dict_row

DSN = os.environ["DATABASE_URL"]

with psycopg.connect(DSN) as conn:
    # 参数化查询：用 %s 占位，值放在第二个参数（一个元组）里
    rows = conn.execute(
        "SELECT id, name, price FROM products WHERE price < %s ORDER BY id",
        (20,),
    ).fetchall()
    for r in rows:
        print(r)

    # 结果按字典返回
    with conn.cursor(row_factory=dict_row) as cur:
        cur.execute("SELECT name, tags FROM products WHERE %s = ANY(tags)", ("kitchen",))
        print(cur.fetchall())

    # 写入并取回自增 id
    new_id = conn.execute(
        "INSERT INTO products (name, price, stock) VALUES (%s, %s, %s) RETURNING id",
        ("Filter Paper", 3.20, 200),
    ).fetchone()[0]
    print("new id:", new_id)
# 离开 with 块：没有异常就 COMMIT，有异常就 ROLLBACK，然后关闭连接

⚠ psycopg 默认自动开启事务

psycopg 在你执行第一条语句时自动 BEGIN，直到你 commit 或离开 with 块才提交。 如果没用 with、也忘了 conn.commit()，写入就不会生效，连接还会一直处于 idle in transaction 状态，挡住 VACUUM（见专题 4）。 另外，不要把脚本命名为 copy.py、json.py 这类和标准库同名的文件，它会挡住标准库模块，psycopg 一导入就报错。

事务

第 2 章的转账例子：两条 UPDATE 必须一起成功或一起失败。psycopg 用 conn.transaction() 包住一段代码：

def transfer(conn, src, dst, amount):
    with conn.transaction():          # BEGIN … COMMIT；块内抛异常则 ROLLBACK
        conn.execute("UPDATE accounts SET balance = balance - %s WHERE id = %s", (amount, src))
        conn.execute("UPDATE accounts SET balance = balance + %s WHERE id = %s", (amount, dst))

with psycopg.connect(DSN, autocommit=True) as conn:
    transfer(conn, 1, 2, 30)
    print(conn.execute("SELECT id, balance FROM accounts ORDER BY id").fetchall())
    try:
        transfer(conn, 1, 2, 500)     # 余额不足，违反 CHECK 约束
    except psycopg.errors.CheckViolation as e:
        print("失败:", e.diag.message_primary, "| sqlstate", e.sqlstate)
    print(conn.execute("SELECT id, balance FROM accounts ORDER BY id").fetchall())

autocommit=True 让连接不再自动 BEGIN，每条语句单独提交，事务的边界完全由 conn.transaction() 决定。这种“显式事务”的写法更不容易出错。

  'SELECT id, name, price, stock FROM products WHERE price < $1 ORDER BY id',
  [20],

// 事务：必须从池里借出【同一个】连接，BEGIN / COMMIT 都在它上面执行
async function transfer(src, dst, amount) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await client.query('UPDATE accounts SET balance = balance - $1 WHERE id = $2', [amount, src]);
    await client.query('UPDATE accounts SET balance = balance + $1 WHERE id = $2', [amount, dst]);
    await client.query('COMMIT');
  } catch (e) {
    await client.query('ROLLBACK');
    throw e;
  } finally {
    client.release();            // 一定要归还，否则池很快被耗尽
  }
}

try {
  await transfer(1, 2, 1000);
} catch (e) {
  console.log('失败:', e.code, e.constraint);   // 失败: 23514 accounts_balance_check
}
await pool.end();

⚠ 事务里不要用 pool.query

pool.query('BEGIN') 和下一句 pool.query('UPDATE …') 可能被分到两个不同的连接上，BEGIN 就白开了。 事务必须 pool.connect() 借出一个 client，所有语句都在它上面执行，最后 release()。

类型的坑：数字变成了字符串

上面输出里 price: '12.50' 是字符串。node-postgres 默认把 numeric 和 bigint（包括 count(*) 的结果）转成字符串，因为 JavaScript 的 number 放不下它们的全部精度：

const r = await pool.query('SELECT count(*) AS n, sum(price) AS total FROM products');
console.log(r.rows[0], typeof r.rows[0].n);
// { n: '4', total: '68.70' } string

处理办法：在 SQL 里转换（count(*)::int），或者在应用里用 Number() / 十进制库处理。金额最好保持字符串或使用 decimal.js 之类的库，不要转成浮点数。

连接池

附录 A.3 讲过：Postgres 为每个连接起一个操作系统进程，开关连接很贵，所以要用连接池复用。node-postgres 的 Pool 自带池；Python 用 psycopg_pool： 

from psycopg_pool import ConnectionPool

pool = ConnectionPool(DSN, min_size=2, max_size=10, open=True)

with pool.connection() as conn:          # 借一个连接，块结束时提交并归还
    conn.execute("SELECT 1")

池应该开多大

直觉上越大越好，其实不是。真正同时干活的连接数受 CPU 核数和磁盘限制，连接太多只会互相争抢。一个常用的起点是 每个数据库服务器 “CPU 核数 × 2 ~ 4” 个活跃连接，再除以应用实例数，得到每个实例的池大小，然后压测调整。 请求多于连接时，让请求在池里排队，比让它们都挤进数据库更快。

PgBouncer 事务模式的限制

PgBouncer 最常用的是事务模式（transaction pooling）：一个客户端只在事务期间占用一个服务器连接，事务一结束连接就给别人。这意味着跨事务的会话状态不可靠：

功能事务模式下替代做法
SET（会话级）✗ 可能被别的客户端继承事务里用 SET LOCAL / set_config(…, true)
会话级咨询锁 pg_advisory_lock✗ 锁留在了服务器连接上用 pg_advisory_xact_lock
LISTEN（第 11 章）✗ 收不到通知监听的那个连接直连 Postgres
临时表✗ 跨事务不可用在一个事务内建、用、删
预备语句（prepared statements）老版本 ✗；PgBouncer 1.21 起可通过 max_prepared_statements 支持升级 PgBouncer 或关闭驱动的预备语句

这正是专题 3 坚持用 SET LOCAL 的原因。配合 RLS 的请求处理函数长这样（连接的是专题 3 的 rlstopic 数据库，输出对应专题 3 前半部分的数据）：

def list_notes(conn, tenant_id):
    with conn.transaction():
        # 租户 ID 来自服务器端验证过的登录信息，而不是请求参数
        conn.execute("SELECT set_config('app.tenant_id', %s, true)", (str(tenant_id),))
        return conn.execute("SELECT id, title FROM notes ORDER BY id").fetchall()

错误处理与超时

按 SQLSTATE 处理错误

Postgres 的每个错误都带一个 5 位的 SQLSTATE 代码。按代码判断，比匹配错误消息文字可靠得多（消息可能随版本、语言设置变化）：

SQLSTATE名称常见原因应用该怎么做
23505unique_violation重复的邮箱、订单号提示用户“已存在”，或改用 ON CONFLICT（专题 2）
23503foreign_key_violation引用了不存在的记录返回 400 / 404
23514check_violation余额为负等返回业务错误
40001serialization_failure可重复读 / 串行化下的并发冲突整个事务重试
40P01deadlock_detected死锁整个事务重试
57014query_canceledstatement_timeout 超时记录慢查询，返回 503
53300too_many_connections连接数超过 max_connections调小池、加 PgBouncer

psycopg 为每个 SQLSTATE 都定义了异常类（psycopg.errors.UniqueViolation 等），node-postgres 则放在 e.code 里。

带重试的串行化事务

专题 2 说过，SERIALIZABLE 事务失败时要重试。下面两个线程同时从同一个账户扣钱（故意“先读、在应用里算、再写回”）：

import time, threading
import psycopg
from psycopg import errors
from psycopg_pool import ConnectionPool

pool = ConnectionPool(DSN, min_size=2, max_size=10, open=True)

def run_serializable(fn, retries=5):
    """在 SERIALIZABLE 下执行 fn(conn)，遇到串行化失败或死锁就整体重试"""
    for attempt in range(1, retries + 1):
        try:
            with pool.connection() as conn:
                conn.isolation_level = psycopg.IsolationLevel.SERIALIZABLE
                with conn.transaction():
                    return fn(conn)
        except (errors.SerializationFailure, errors.DeadlockDetected) as e:
            print(f"  第 {attempt} 次失败（{e.sqlstate}），重试")
            time.sleep(0.05 * attempt)            # 稍等再试，避免再次撞车
    raise RuntimeError("重试次数用完")

def withdraw(conn, amount):
    bal = conn.execute("SELECT balance FROM accounts WHERE id = 1").fetchone()[0]
    time.sleep(0.2)                               # 模拟耗时的业务逻辑
    conn.execute("UPDATE accounts SET balance = %s WHERE id = 1", (bal - amount,))
    return bal - amount

# 余额 100，两个线程分别扣 10 和 20
threads = [threading.Thread(target=lambda a=a: print("结果", run_serializable(lambda c: withdraw(c, a))))
           for a in (10, 20)]
for t in threads: t.start()
for t in threads: t.join()

结果 80
  第 1 次失败（40001），重试
结果 70
最终余额: 70          ✓ 没有丢失更新
★ 重试的是整个事务

重试时必须从头重新读数据再计算，所以重试的单位是包含读取在内的整个函数，而不是失败的那一条语句。也因此，事务里不要做“发邮件、调第三方支付”这类无法撤销的事。

超时

一条失控的慢查询会一直占着连接。在连接参数里给所有语句设上限，并用 application_name 标明身份，方便在 pg_stat_activity 里认出它：

with psycopg.connect(DSN, application_name="shop-api",
                     options="-c statement_timeout=1000") as conn:   # 1 秒
    try:
        conn.execute("SELECT pg_sleep(3)")
    except psycopg.errors.QueryCanceled as e:
        print("超时:", e.diag.message_primary, e.sqlstate)

超时: canceling statement due to statement timeout 57014

node-postgres 的对应写法是在 Pool 配置里加 statement_timeout: 1000 和 application_name: 'shop-api'。也可以对整个角色设置：ALTER ROLE app_user SET statement_timeout = '5s';。

批量写入：用 COPY

一次导入成千上万行时，逐条 INSERT 很慢。Postgres 的 COPY 协议专门用于批量传输，驱动都支持。插入 10 万行对比：

rows = [(f"item {i}", i % 100 + 0.99, i % 50) for i in range(100_000)]

with conn.cursor() as cur:
    cur.executemany("INSERT INTO t1 VALUES (%s, %s, %s)", rows)

with conn.cursor() as cur:
    with cur.copy("COPY t2 (name, price, stock) FROM STDIN") as copy:
        for r in rows:
            copy.write_row(r)

executemany: 1.82s
COPY:        0.09s       约快 20 倍

Node 里可以用 pg-copy-streams 包实现同样的事。几百行以内的小批量，用一条多行 INSERT … VALUES (…), (…), … 或 INSERT … SELECT * FROM unnest($1::text[], $2::numeric[]) 也足够快。

ORM 与其他语言

ORM（对象关系映射）把表映射成类，帮你生成 SQL、管理迁移，能省很多样板代码。它们底层用的仍是上面这些驱动。用 ORM 时有两点要注意：

  • 看得见它生成的 SQL：开发时打开 ORM 的 SQL 日志。一个页面发了几十条查询，多半是 N+1 问题。
  • Postgres 特色功能可以直接写 SQL：JSONB 运算符、全文搜索、pgvector、窗口函数、ON CONFLICT，ORM 支持不全时就用它提供的“原生 SQL”接口，同样记得参数化。

小结

  • 连接串 postgresql://用户:密码@主机:端口/数据库 放在环境变量里，不进代码仓库。
  • 任何来自用户的值都用参数传（%s、$1），绝不拼进 SQL。
  • 事务要么用 psycopg 的 conn.transaction()，要么在 node-postgres 里借出同一个 client 执行 BEGIN / COMMIT / ROLLBACK 并 release。
  • 用连接池；所有实例的池加起来要小于 max_connections。PgBouncer 事务模式下不要依赖会话级状态。
  • 按 SQLSTATE 处理错误；40001 和 40P01 要重试整个事务；给语句设 statement_timeout。
  • 批量导入用 COPY，比逐条 INSERT 快一个数量级。
  • 注意类型映射：numeric 在 Python 里是 Decimal，在 node-postgres 里是字符串。

  3. 把本页的 run_serializable 改成 node-postgres 版本，用两个并发的 Promise 测试它。
    参考答案 
    import pg from 'pg';
const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL, max: 5 });
const sleep = ms => new Promise(r => setTimeout(r, ms));

async function runSerializable(fn, retries = 5) {
  for (let attempt = 1; attempt <= retries; attempt++) {
    const client = await pool.connect();
    try {
      await client.query('BEGIN ISOLATION LEVEL SERIALIZABLE');
      const result = await fn(client);
      await client.query('COMMIT');
      return result;
    } catch (e) {
      await client.query('ROLLBACK');
      if ((e.code === '40001' || e.code === '40P01') && attempt < retries) {
        console.log(`  第 ${attempt} 次失败（${e.code}），重试`);
        await sleep(50 * attempt);
        continue;
      }
      throw e;
    } finally {
      client.release();
    }
  }
}

async function withdraw(client, amount) {
  const { rows } = await client.query('SELECT balance FROM accounts WHERE id = 1');

  const next = Number(rows[0].balance) - amount;      // numeric 返回的是字符串
  await client.query('UPDATE accounts SET balance = $1 WHERE id = 1', [next]);
  return next;
}

const results = await Promise.all([10, 20].map(a => runSerializable(c => withdraw(c, a))));

  4. 把 Python 连接池的 max_size 设为 2，同时启动 5 个线程各执行 SELECT pg_sleep(1)，观察总耗时，理解“请求在池里排队”。
    参考答案 
    import time, threading
from psycopg_pool import ConnectionPool

with ConnectionPool(DSN, min_size=2, max_size=2, open=True) as pool:
    pool.wait()
    def work():
        with pool.connection() as c:
            c.execute("SELECT pg_sleep(1)")
    t = time.perf_counter()
    threads = [threading.Thread(target=work) for _ in range(5)]
    for x in threads: x.start()
    for x in threads: x.join()
    print(f"{time.perf_counter() - t:.2f} s")      # 3.01 s

    5 个请求只有 2 个连接：第一批 2 个、第二批 2 个、最后 1 个，各 1 秒，总共约 3 秒。请求在池里排队，数据库同时只处理 2 个查询。等待超过 timeout（默认 30 秒）还借不到连接，psycopg_pool 会抛出 PoolTimeout。

  5. 用 COPY 把一个 CSV 文件导入 Postgres，并和逐行 INSERT 比较耗时。
    参考答案 
    import csv, time, psycopg

with psycopg.connect(DSN) as conn:
    conn.execute("CREATE TABLE items_copy (name text, price numeric, stock int)")
    conn.execute("CREATE TABLE items_insert (name text, price numeric, stock int)")

    t = time.perf_counter()
    with open('items.csv') as f, conn.cursor() as cur:
        with cur.copy("COPY items_copy (name, price, stock) FROM STDIN WITH (FORMAT csv, HEADER)") as copy:
            while data := f.read(65536):
                copy.write(data)                    # 原样把文件内容流给服务器
    print(f"COPY:        {time.perf_counter() - t:.2f} s")

    t = time.perf_counter()
    with open('items.csv') as f, conn.cursor() as cur:
        for row in list(csv.reader(f))[1:]:
            cur.execute("INSERT INTO items_insert VALUES (%s, %s, %s)", row)
    print(f"逐行 INSERT: {time.perf_counter() - t:.2f} s")

    COPY:        0.04 s
逐行 INSERT: 7.20 s       （10 万行的 CSV）

    逐行 INSERT 每一行都要一次网络往返；COPY 一次把整个文件流过去，由服务器批量解析写入。在 psql 里也可以直接 \copy items_copy FROM 'items.csv' CSV HEADER。
